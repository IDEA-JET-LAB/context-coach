# Story 14.5-5: Data Minimization Pipeline

Status: Ready for Dev
Priority: P1

## Story

**As a** privacy-conscious user,
**I want** only essential data stored,
**So that** my full response history and file paths are not retained unnecessarily.

## Acceptance Criteria

1. **Given** a response is captured
   **When** it exceeds 1000 characters
   **Then** a summary is generated instead of storing full text
   **And** the original is discarded after summarization

2. **Given** a file path in captured data
   **When** data minimization is enabled
   **Then** the path is hashed (e.g., `/Users/john/project/file.ts` becomes `path:abc123`)
   **And** only the filename is preserved if needed

3. **Given** tool input data is captured
   **When** input exceeds 500 characters
   **Then** it is truncated with `[TRUNCATED]` marker
   **And** the full input is not stored

4. **Given** user's capture level is set to "minimal"
   **When** a prompt is captured
   **Then** only analysis results are stored
   **And** no raw prompt text is kept

5. **Given** user's capture level is set to "local"
   **When** any capture occurs
   **Then** data is stored only locally
   **And** nothing is synced to cloud

6. **Given** a capture contains large file contents
   **When** data minimization is applied
   **Then** file contents are replaced with metadata only
   **And** format is `[FILE: filename, SIZE: Xkb, TYPE: extension]`

7. **Given** input text is empty, null, or undefined
   **When** minimization is called
   **Then** the function returns gracefully without errors
   **And** metadata reflects zero-length input

8. **Given** AI summarization fails (network error, timeout, API error)
   **When** a long response requires summarization
   **Then** the system falls back to truncation (first 800 + last 200 chars)
   **And** the result includes marker `[SUMMARIZATION FAILED - TRUNCATED]`
   **And** the failure is logged for monitoring

### Minimization Pipeline

```typescript
// lib/capture/minimize-data.ts

import { summarizeResponse } from './summarize-response';
import { hashFilePaths } from './hash-paths';
import { truncateToolInput } from './truncate-content';
import { PrivacySettings } from '@/lib/privacy/types';

export interface MinimizationOptions {
  captureLevel: 'full' | 'standard' | 'minimal' | 'local';
  hashFilePaths: boolean;
  responseSummaryThreshold: number;  // default: 1000
  toolInputTruncateThreshold: number; // default: 500
}

export interface MinimizationResult {
  minimizedPrompt: string | null;
  minimizedResponse: string | null;
  toolInputs: Record<string, string>;
  metadata: {
    originalPromptLength: number;
    originalResponseLength: number;
    wasResponseSummarized: boolean;
    pathsHashed: number;
    truncations: number;
  };
}

export async function minimizeData(
  prompt: string | null | undefined,
  response: string | null | undefined,
  toolInputs: Record<string, string> | null | undefined,
  options: MinimizationOptions
): Promise<MinimizationResult> {
  // Handle empty/null/undefined inputs gracefully
  const safePrompt = prompt ?? '';
  const safeResponse = response ?? null;
  const safeToolInputs = toolInputs ?? {};

  const metadata = {
    originalPromptLength: safePrompt.length,
    originalResponseLength: safeResponse?.length || 0,
    wasResponseSummarized: false,
    pathsHashed: 0,
    truncations: 0,
  };

  // Early return for empty prompt (nothing to minimize)
  if (safePrompt.length === 0 && !safeResponse && Object.keys(safeToolInputs).length === 0) {
    return {
      minimizedPrompt: null,
      minimizedResponse: null,
      toolInputs: {},
      metadata,
    };
  }

  // Handle capture levels
  if (options.captureLevel === 'minimal') {
    // Only analysis, no raw text
    return {
      minimizedPrompt: null,
      minimizedResponse: null,
      toolInputs: {},
      metadata,
    };
  }

  if (options.captureLevel === 'local') {
    // Will be handled by capture hook - skip cloud sync
    // Still minimize for local storage
  }

  // 1. Hash file paths if enabled
  let processedPrompt = safePrompt;
  let processedResponse = safeResponse;

  if (options.hashFilePaths) {
    const promptResult = hashFilePaths(processedPrompt);
    processedPrompt = promptResult.text;
    metadata.pathsHashed += promptResult.pathCount;

    if (processedResponse) {
      const responseResult = hashFilePaths(processedResponse);
      processedResponse = responseResult.text;
      metadata.pathsHashed += responseResult.pathCount;
    }
  }

  // 2. Summarize long responses
  if (processedResponse && processedResponse.length > options.responseSummaryThreshold) {
    processedResponse = await summarizeResponse(processedResponse);
    metadata.wasResponseSummarized = true;
  }

  // 3. Truncate tool inputs
  const processedToolInputs: Record<string, string> = {};
  for (const [tool, input] of Object.entries(safeToolInputs)) {
    if (input.length > options.toolInputTruncateThreshold) {
      processedToolInputs[tool] = truncateToolInput(input, options.toolInputTruncateThreshold);
      metadata.truncations++;
    } else {
      processedToolInputs[tool] = input;
    }
  }

  // 4. Replace file contents
  processedPrompt = replaceFileContents(processedPrompt);
  if (processedResponse) {
    processedResponse = replaceFileContents(processedResponse);
  }

  return {
    minimizedPrompt: processedPrompt,
    minimizedResponse: processedResponse,
    toolInputs: processedToolInputs,
    metadata,
  };
}
```

### Response Summarization

```typescript
// lib/capture/summarize-response.ts

import OpenAI from 'openai';

const openai = new OpenAI();

/**
 * Fallback truncation when AI summarization fails.
 * Keeps first 800 and last 200 characters to preserve context.
 */
function fallbackTruncate(response: string): string {
  const keepStart = 800;
  const keepEnd = 200;
  const truncated = `${response.slice(0, keepStart)}\n[SUMMARIZATION FAILED - TRUNCATED: ${response.length - keepStart - keepEnd} chars removed]\n${response.slice(-keepEnd)}`;
  return truncated;
}

export async function summarizeResponse(response: string): Promise<string> {
  try {
    const result = await openai.chat.completions.create({
      model: 'gpt-4o-mini',
      messages: [
        {
          role: 'system',
          content: `Summarize this AI response in 2-3 sentences.
Focus on: what was done, key decisions made, and outcomes.
Do NOT include any code, file paths, or sensitive information.`,
        },
        {
          role: 'user',
          content: response.slice(0, 4000), // Limit input size
        },
      ],
      max_tokens: 200,
      temperature: 0.3,
    });

    const summary = result.choices[0]?.message?.content || '[Summary unavailable]';

    return `[SUMMARIZED: ${response.length} chars -> ${summary.length} chars]\n${summary}`;
  } catch (error) {
    // Log failure for monitoring
    console.error('[Summarization Failed]', {
      error: error instanceof Error ? error.message : 'Unknown error',
      responseLength: response.length,
      timestamp: new Date().toISOString(),
    });

    // Fallback to truncation
    return fallbackTruncate(response);
  }
}
```

### File Path Hashing

**Unicode Normalization:** File paths must be normalized to NFC (Canonical Decomposition, followed by Canonical Composition) before hashing to ensure consistent deduplication. For example, `é` can be represented as a single code point (U+00E9) or as `e` + combining accent (U+0065 + U+0301). Without normalization, the same visual path would produce different hashes.

```typescript
// lib/capture/hash-paths.ts

import crypto from 'crypto';

export interface PathHashResult {
  text: string;
  pathCount: number;
  pathMap: Record<string, string>; // For debugging/lookup
}

// Match absolute paths with 4+ segments
const PATH_PATTERN = /(?:\/[a-zA-Z0-9_.-]+){4,}/g;

export function hashFilePaths(text: string, preserveFilename = true): PathHashResult {
  const pathMap: Record<string, string> = {};
  let pathCount = 0;

  const result = text.replace(PATH_PATTERN, (match) => {
    pathCount++;

    // Normalize Unicode to NFC before hashing for consistent deduplication
    const normalizedPath = match.normalize('NFC');
    const hash = crypto.createHash('sha256').update(normalizedPath).digest('hex').slice(0, 8);

    if (preserveFilename) {
      const filename = match.split('/').pop();
      pathMap[match] = `[PATH:${hash}]/${filename}`;
      return pathMap[match];
    }

    pathMap[match] = `[PATH:${hash}]`;
    return pathMap[match];
  });

  return { text: result, pathCount, pathMap };
}
```

### Tool Input Truncation

```typescript
// lib/capture/truncate-content.ts

export function truncateToolInput(input: string, maxLength: number): string {
  if (input.length <= maxLength) return input;

  const keepStart = Math.floor(maxLength * 0.4);
  const keepEnd = Math.floor(maxLength * 0.3);
  const removed = input.length - keepStart - keepEnd;

  return `${input.slice(0, keepStart)}\n[TRUNCATED: ${removed} chars removed]\n${input.slice(-keepEnd)}`;
}
```

### File Content Replacement

```typescript
// lib/capture/minimize-data.ts (continued)

const FILE_CONTENT_PATTERNS = [
  // Match "File: path" followed by content
  /(?:File|Reading|Content of)[:\s]+([^\n]+)\n```[\s\S]*?```/gi,
  // Match inline file dumps
  /(?:```(?:typescript|javascript|python|json|yaml|tsx|jsx)[^\n]*\n)([\s\S]{500,}?)(?:```)/gi,
];

export function replaceFileContents(text: string): string {
  let result = text;

  // Replace file content blocks with metadata
  result = result.replace(
    /(?:File|Reading)[:\s]+([^\n]+)\n```[\s\S]*?```/gi,
    (match, filename) => {
      const extension = filename.split('.').pop() || 'unknown';
      const size = Math.round(match.length / 1024);
      return `[FILE: ${filename}, SIZE: ${size}kb, TYPE: ${extension}]`;
    }
  );

  // Replace large inline code blocks
  result = result.replace(
    /```(?:typescript|javascript|python|json|yaml|tsx|jsx)?[^\n]*\n([\s\S]{1000,}?)```/gi,
    (match) => {
      const size = Math.round(match.length / 1024);
      return `[CODE BLOCK: ${size}kb, see full version in IDE]`;
    }
  );

  return result;
}
```

### Integration with Capture Pipeline

```typescript
// lib/capture/pipeline.ts

import { redactSensitiveData } from './redact-enhanced';
import { minimizeData } from './minimize-data';
import { getPrivacyPreferences } from '@/lib/privacy/preferences';

export async function processCapturedData(
  userId: string,
  rawPrompt: string,
  rawResponse: string | null,
  toolInputs: Record<string, string>
): Promise<ProcessedCapture> {
  // 1. Get user preferences
  const preferences = await getPrivacyPreferences(userId);

  // 2. Redact sensitive data FIRST
  const redactionResult = redactSensitiveData(rawPrompt, {
    redactFilePaths: preferences.redactFilePaths,
    redactDbQueries: true,
    customPatterns: preferences.customPatterns,
  });

  // 3. Then minimize data
  const minimizationResult = await minimizeData(
    redactionResult.redactedText,
    rawResponse,
    toolInputs,
    {
      captureLevel: preferences.captureLevel,
      hashFilePaths: preferences.hashFilePaths,
      responseSummaryThreshold: 1000,
      toolInputTruncateThreshold: 500,
    }
  );

  // 4. Return processed capture
  return {
    prompt: minimizationResult.minimizedPrompt,
    response: minimizationResult.minimizedResponse,
    toolInputs: minimizationResult.toolInputs,
    redactionStats: redactionResult.stats,
    minimizationStats: minimizationResult.metadata,
    shouldSync: preferences.captureLevel !== 'local',
  };
}
```
