# Story 14.5-6: Privacy Preferences Database

Status: Ready for Dev
Priority: P0

## Story

**As a** system,
**I want** a dedicated table to store user privacy preferences,
**So that** all privacy controls are persisted and applied consistently across the platform.

## Acceptance Criteria

1. **Given** a new user signs up
   **When** their account is created
   **Then** a privacy_preferences row is created with default values

2. **Given** the privacy_preferences table
   **When** queried
   **Then** it includes columns for:
   - Capture level (full, standard, minimal, local)
   - Redaction toggles (file paths, emails, DB queries)
   - Custom redaction patterns
   - Excluded projects
   - Retention period
   - Consent timestamps
   - Capture pause status

3. **Given** RLS policies on privacy_preferences
   **When** a user queries the table
   **Then** they can only see/modify their own preferences

4. **Given** another user's preferences
   **When** I try to read or update them
   **Then** the operation is denied by RLS

5. **Given** the capture system needs preferences
   **When** it queries for a user
   **Then** defaults are returned if no row exists

6. **Given** a user updates preferences via the API
   **When** the update completes
   **Then** `updated_at` timestamp is refreshed
   **And** the change is logged for audit

7. **Given** a user adds custom regex patterns
   **When** the patterns are validated
   **Then** invalid regex syntax is rejected with a helpful error message
   **And** regex complexity is limited to prevent ReDoS attacks (max 100 chars, no nested quantifiers)
   **And** each pattern is tested against sample text before saving

```sql
-- Migration: YYYYMMDDHHMMSS_privacy_preferences.sql
-- NOTE: Use actual timestamp when creating the migration file

-- Create privacy preferences table
CREATE TABLE privacy_preferences (
  -- Primary key - one row per user
  user_id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,

  -- Capture level control
  capture_level TEXT NOT NULL DEFAULT 'standard'
    CHECK (capture_level IN ('full', 'standard', 'minimal', 'local')),

  -- Redaction toggles
  redact_file_paths BOOLEAN NOT NULL DEFAULT TRUE,
  redact_emails BOOLEAN NOT NULL DEFAULT TRUE,
  redact_db_queries BOOLEAN NOT NULL DEFAULT TRUE,

  -- Custom patterns (regex strings)
  custom_patterns TEXT[] NOT NULL DEFAULT '{}',

  -- Excluded projects (project paths or IDs)
  excluded_projects TEXT[] NOT NULL DEFAULT '{}',

  -- Retention settings
  retention_days INTEGER DEFAULT 90
    CHECK (retention_days IS NULL OR retention_days >= 0),

  -- Capture pause
  capture_paused BOOLEAN NOT NULL DEFAULT FALSE,
  capture_paused_at TIMESTAMPTZ,

  -- Consent tracking
  consent_given_at TIMESTAMPTZ,
  consent_last_reviewed_at TIMESTAMPTZ,

  -- Granular consent (from Phase 2)
  capture_prompts BOOLEAN NOT NULL DEFAULT TRUE,
  capture_responses BOOLEAN NOT NULL DEFAULT FALSE,
  capture_sessions BOOLEAN NOT NULL DEFAULT TRUE,
  capture_tool_usage BOOLEAN NOT NULL DEFAULT TRUE,

  -- Analysis consent
  enable_scoring BOOLEAN NOT NULL DEFAULT TRUE,
  enable_pattern_analysis BOOLEAN NOT NULL DEFAULT TRUE,
  enable_team_comparison BOOLEAN NOT NULL DEFAULT TRUE,

  -- Sharing consent
  share_with_team BOOLEAN NOT NULL DEFAULT TRUE,
  include_in_benchmarks BOOLEAN NOT NULL DEFAULT FALSE,
  allow_ai_training BOOLEAN NOT NULL DEFAULT FALSE,

  -- Timestamps
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Create index for lookup
CREATE INDEX idx_privacy_preferences_user_id ON privacy_preferences(user_id);

-- Enable RLS
ALTER TABLE privacy_preferences ENABLE ROW LEVEL SECURITY;

-- RLS Policies

-- Users can read their own preferences
CREATE POLICY "Users can read own privacy preferences"
  ON privacy_preferences
  FOR SELECT
  USING (auth.uid() = user_id);

-- Users can update their own preferences
CREATE POLICY "Users can update own privacy preferences"
  ON privacy_preferences
  FOR UPDATE
  USING (auth.uid() = user_id)
  WITH CHECK (auth.uid() = user_id);

-- Users can insert their own preferences (for edge cases)
CREATE POLICY "Users can insert own privacy preferences"
  ON privacy_preferences
  FOR INSERT
  WITH CHECK (auth.uid() = user_id);

-- Service role can access all (for capture system)
CREATE POLICY "Service role can access all privacy preferences"
  ON privacy_preferences
  FOR ALL
  USING (auth.jwt() ->> 'role' = 'service_role');

-- Updated_at trigger
CREATE OR REPLACE FUNCTION update_privacy_preferences_timestamp()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER privacy_preferences_updated_at
  BEFORE UPDATE ON privacy_preferences
  FOR EACH ROW EXECUTE FUNCTION update_privacy_preferences_timestamp();

-- Auto-create preferences for new users
CREATE OR REPLACE FUNCTION create_default_privacy_preferences()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO public.privacy_preferences (user_id)
  VALUES (NEW.id)
  ON CONFLICT (user_id) DO NOTHING;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE TRIGGER create_privacy_preferences_on_signup
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION create_default_privacy_preferences();
```

### React Hook

```typescript
// lib/hooks/use-privacy-preferences.ts

'use client';

import { useState, useEffect } from 'react';
import { createBrowserClient } from '@/lib/supabase/client';
import { PrivacyPreferences, DEFAULT_PRIVACY_PREFERENCES } from '@/lib/privacy/types';
import { toast } from 'sonner';

export function usePrivacyPreferences() {
  const [preferences, setPreferences] = useState<PrivacyPreferences | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState<Error | null>(null);

  const supabase = createBrowserClient();

  useEffect(() => {
    async function fetchPreferences() {
      try {
        const { data: { user } } = await supabase.auth.getUser();
        if (!user) throw new Error('Not authenticated');

        const { data, error } = await supabase
          .from('privacy_preferences')
          .select('*')
          .eq('user_id', user.id)
          .single();

        if (error && error.code !== 'PGRST116') {
          throw error;
        }

        // If no row exists, use defaults
        if (!data) {
          setPreferences({
            ...DEFAULT_PRIVACY_PREFERENCES,
            userId: user.id,
            createdAt: new Date().toISOString(),
            updatedAt: new Date().toISOString(),
          } as PrivacyPreferences);
        } else {
          setPreferences(mapDbRowToPreferences(data));
        }
      } catch (err) {
        setError(err instanceof Error ? err : new Error('Failed to load preferences'));
      } finally {
        setIsLoading(false);
      }
    }

    fetchPreferences();
  }, [supabase]);

  const updatePreferences = async (updates: Partial<PrivacyPreferences>) => {
    if (!preferences) return;

    try {
      const { error } = await supabase
        .from('privacy_preferences')
        .upsert({
          user_id: preferences.userId,
          ...mapPreferencesToDbRow(updates),
        });

      if (error) throw error;

      setPreferences({ ...preferences, ...updates, updatedAt: new Date().toISOString() });
      toast.success('Preferences saved');
    } catch (err) {
      toast.error('Failed to save preferences');
      throw err;
    }
  };

  const hasConsented = preferences?.consentGivenAt !== null;

  const saveConsent = async (consentPreferences: Partial<PrivacyPreferences>) => {
    await updatePreferences({
      ...consentPreferences,
      consentGivenAt: new Date().toISOString(),
      consentLastReviewedAt: new Date().toISOString(),
    });
  };

  return {
    preferences,
    isLoading,
    error,
    hasConsented,
    updatePreferences,
    saveConsent,
  };
}

    consentGivenAt: row.consent_given_at as string | null,
    consentLastReviewedAt: row.consent_last_reviewed_at as string | null,
    capturePrompts: row.capture_prompts as boolean,
    captureResponses: row.capture_responses as boolean,
    captureSessions: row.capture_sessions as boolean,
    captureToolUsage: row.capture_tool_usage as boolean,
```

### Server-Side Access

```typescript
// lib/privacy/server.ts

import { createServiceRoleClient } from '@/lib/supabase/service-role';
import { PrivacyPreferences, DEFAULT_PRIVACY_PREFERENCES } from './types';

export async function getPrivacyPreferences(userId: string): Promise<PrivacyPreferences> {
  const supabase = createServiceRoleClient();

  const { data, error } = await supabase
    .from('privacy_preferences')
    .select('*')
    .eq('user_id', userId)
    .single();

  if (error && error.code !== 'PGRST116') {
    throw error;
  }

  if (!data) {
    return {
      ...DEFAULT_PRIVACY_PREFERENCES,
      userId,
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString(),
    } as PrivacyPreferences;
  }

  return mapDbRowToPreferences(data);
}

export async function updatePrivacyPreferences(
  userId: string,
  updates: Partial<PrivacyPreferences>
): Promise<void> {
  const supabase = createServiceRoleClient();

  const { error } = await supabase
    .from('privacy_preferences')
    .upsert({
      user_id: userId,
      ...mapPreferencesToDbRow(updates),
    });

  if (error) throw error;
}
```

### Audit Logging

```sql
-- Create audit log for preference changes
CREATE TABLE privacy_preferences_audit (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  changed_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  changed_fields JSONB NOT NULL,
  old_values JSONB NOT NULL,
  new_values JSONB NOT NULL
);

-- Trigger to log changes
CREATE OR REPLACE FUNCTION log_privacy_preferences_changes()
RETURNS TRIGGER AS $$
DECLARE
  changed_fields JSONB := '{}';
  old_values JSONB := '{}';
  new_values JSONB := '{}';
BEGIN
  -- Check each field for changes (excluding custom_patterns for privacy)
  IF OLD.capture_level IS DISTINCT FROM NEW.capture_level THEN
    changed_fields := changed_fields || '{"capture_level": true}';
    old_values := old_values || jsonb_build_object('capture_level', OLD.capture_level);
    new_values := new_values || jsonb_build_object('capture_level', NEW.capture_level);
  END IF;

  IF OLD.retention_days IS DISTINCT FROM NEW.retention_days THEN
    changed_fields := changed_fields || '{"retention_days": true}';
    old_values := old_values || jsonb_build_object('retention_days', OLD.retention_days);
    new_values := new_values || jsonb_build_object('retention_days', NEW.retention_days);
  END IF;

  -- Add more fields as needed...

  -- Only log if something changed
  IF changed_fields != '{}' THEN
    INSERT INTO privacy_preferences_audit (user_id, changed_fields, old_values, new_values)
    VALUES (NEW.user_id, changed_fields, old_values, new_values);
  END IF;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE TRIGGER privacy_preferences_audit_trigger
  AFTER UPDATE ON privacy_preferences
  FOR EACH ROW EXECUTE FUNCTION log_privacy_preferences_changes();
```
