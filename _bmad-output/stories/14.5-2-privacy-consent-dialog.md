# Story 14.5-2: Privacy Consent Dialog

Status: Ready for Dev
Priority: P0
Dependencies: Story 14.5-6 (Privacy Preferences Database) must be implemented first

## Story

**As a** new Contextor user,
**I want** to understand what data is collected and give my consent,
**So that** I can make an informed decision before my data is captured.

## Acceptance Criteria

1. **Given** I am a new user who has never given consent
   **When** I access the Contextor dashboard for the first time
   **Then** I see a privacy consent dialog that blocks other actions

2. **Given** the consent dialog is displayed
   **When** I view it
   **Then** I see clear explanations of what data is collected:
   - Prompts to Claude
   - Claude's responses (optional)
   - Tool usage information
   - Session timing data

3. **Given** the consent dialog is displayed
   **When** I view it
   **Then** I see what is automatically redacted:
   - API keys and tokens
   - Passwords and secrets
   - SSH private keys
   - Database credentials

4. **Given** the consent dialog
   **When** I click "Configure"
   **Then** I see granular options for each data type
   **And** I can toggle each category on/off

5. **Given** I have made my selections
   **When** I click "I Understand"
   **Then** my consent preferences are saved
   **And** the timestamp is recorded
   **And** the dialog closes

6. **Given** I want to review my consent
   **When** I navigate to Privacy Settings
   **Then** I can view and update my consent preferences
   **And** I see when I last reviewed them

7. **Given** I have previously given consent
   **When** I revoke consent for a data type (e.g., disable prompt capture)
   **Then** future data of that type is no longer collected
   **And** existing data remains (deletion handled separately in Story 14.5-3)
   **And** a confirmation message explains the change

8. **Given** I am saving my consent preferences
   **When** a network or server error occurs
   **Then** I see a clear error message
   **And** my selections remain in the dialog (not lost)
   **And** I can retry saving

## Dev Notes

### Language Guidelines (Critical)

**Avoid surveillance-like language** per PRD requirements. The consent dialog should use empowering, user-centric language that emphasizes:
- User control and choice
- Benefits to the user (learning, improvement)
- Transparency about what happens with data

| Avoid | Prefer |
|-------|--------|
| "We track your prompts" | "Capture prompts you choose to save" |
| "Monitoring your sessions" | "Session insights for your review" |
| "Data collection" (as header) | "Your Privacy Preferences" |
| "We analyze your behavior" | "Analysis to help you improve" |

### Consent Types Interface

```typescript
// lib/privacy/consent.ts

export interface DataConsentPreferences {
  // Core data types
  capturePrompts: boolean;
  captureResponses: boolean;
  captureSessions: boolean;
  captureToolUsage: boolean;

  // Analysis types
  enableScoring: boolean;
  enablePatternAnalysis: boolean;
  enableTeamComparison: boolean;

  // Data sharing
  shareWithTeam: boolean;
  includeInBenchmarks: boolean;
  allowAITraining: boolean; // Always false by default

  // Timestamps
  consentGivenAt: string | null;
  lastReviewedAt: string | null;
}

export const DEFAULT_CONSENT: DataConsentPreferences = {
  capturePrompts: true,
  captureResponses: false,  // Opt-in for responses
  captureSessions: true,
  captureToolUsage: true,
  enableScoring: true,
  enablePatternAnalysis: true,
  enableTeamComparison: true,
  shareWithTeam: true,
  includeInBenchmarks: false,
  allowAITraining: false,  // Always false by default
  consentGivenAt: null,
  lastReviewedAt: null,
};
```

### Consent Dialog Component

```tsx
// components/privacy/consent-dialog.tsx
'use client';

import { useState } from 'react';
import {
  Dialog,
  DialogContent,
  DialogDescription,
  DialogFooter,
  DialogHeader,
  DialogTitle,
} from '@/components/ui/dialog';
import { Button } from '@/components/ui/button';
import { Checkbox } from '@/components/ui/checkbox';
import { Shield, ChevronDown, ChevronUp } from 'lucide-react';
import { DEFAULT_CONSENT, DataConsentPreferences } from '@/lib/privacy/consent';

interface ConsentDialogProps {
  open: boolean;
  onConsent: (preferences: DataConsentPreferences) => void;
}

export function ConsentDialog({ open, onConsent }: ConsentDialogProps) {
  const [preferences, setPreferences] = useState(DEFAULT_CONSENT);
  const [showAdvanced, setShowAdvanced] = useState(false);

  const handleConsent = () => {
    onConsent({
      ...preferences,
      consentGivenAt: new Date().toISOString(),
      lastReviewedAt: new Date().toISOString(),
    });
  };

  return (
    <Dialog open={open}>
      <DialogContent className="max-w-lg">
        <DialogHeader>
          <div className="flex items-center gap-2">
            <Shield className="h-5 w-5 text-primary" />
            <DialogTitle>Data Collection Consent</DialogTitle>
          </div>
          <DialogDescription>
            To provide prompt analysis, we collect the following:
          </DialogDescription>
        </DialogHeader>

        <div className="space-y-4">
          {/* Data collection items */}
          <div className="space-y-2">
            <ConsentItem
              label="Your prompts to Claude"
              checked={preferences.capturePrompts}
              onChange={(v) => setPreferences({ ...preferences, capturePrompts: v })}
            />
            <ConsentItem
              label="Claude's responses (for context)"
              checked={preferences.captureResponses}
              onChange={(v) => setPreferences({ ...preferences, captureResponses: v })}
            />
            <ConsentItem
              label="Tool usage (which tools were called)"
              checked={preferences.captureToolUsage}
              onChange={(v) => setPreferences({ ...preferences, captureToolUsage: v })}
            />
            <ConsentItem
              label="Session timing (duration, frequency)"
              checked={preferences.captureSessions}
              onChange={(v) => setPreferences({ ...preferences, captureSessions: v })}
            />
          </div>

          {/* Automatic redaction notice */}
          <div className="bg-muted rounded-lg p-3">
            <p className="text-sm font-medium mb-2">We automatically redact:</p>
            <ul className="text-sm text-muted-foreground space-y-1">
              <li>* API keys, tokens, passwords</li>
              <li>* SSH private keys</li>
              <li>* Database credentials</li>
            </ul>
          </div>

          {/* Advanced options toggle */}
          <Button
            variant="ghost"
            size="sm"
            onClick={() => setShowAdvanced(!showAdvanced)}
          >
            {showAdvanced ? <ChevronUp /> : <ChevronDown />}
            Advanced Options
          </Button>

          {showAdvanced && (
            <div className="space-y-2 pl-4">
              <ConsentItem
                label="Enable AI-powered scoring"
                checked={preferences.enableScoring}
                onChange={(v) => setPreferences({ ...preferences, enableScoring: v })}
              />
              <ConsentItem
                label="Enable pattern analysis"
                checked={preferences.enablePatternAnalysis}
                onChange={(v) => setPreferences({ ...preferences, enablePatternAnalysis: v })}
              />
              <ConsentItem
                label="Compare with team members"
                checked={preferences.enableTeamComparison}
                onChange={(v) => setPreferences({ ...preferences, enableTeamComparison: v })}
              />
            </div>
          )}
        </div>

        <DialogFooter className="flex-col sm:flex-row gap-2">
          <Button variant="outline" asChild>
            <a href="/privacy" target="_blank">View Privacy Policy</a>
          </Button>
          <Button onClick={handleConsent}>I Understand</Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  );
}
```

### Dashboard Integration

```tsx
// app/(dashboard)/layout.tsx - Add consent check
import { ConsentDialog } from '@/components/privacy/consent-dialog';
import { useConsentPreferences } from '@/lib/hooks/use-consent-preferences';

export default function DashboardLayout({ children }) {
  const { hasConsented, saveConsent, isLoading } = useConsentPreferences();

  if (isLoading) return <LoadingSpinner />;

  return (
    <>
      <ConsentDialog
        open={!hasConsented}
        onConsent={saveConsent}
      />
      {hasConsented && children}
    </>
  );
}
```
